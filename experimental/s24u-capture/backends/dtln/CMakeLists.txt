cmake_minimum_required(VERSION 3.16)
project(clarifae_dtln C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(DTLN_ROOT "${CMAKE_CURRENT_LIST_DIR}")
include("${DTLN_ROOT}/cmake/ClarifaeDTLN.cmake")
add_executable(dtln_probe tests/probe.c ../../engine/backend_dtln.c ../../engine/resampler.c)
target_compile_definitions(dtln_probe PRIVATE CLARIFAE_WITH_DTLN=1)
target_include_directories(dtln_probe PRIVATE ../../engine)
target_link_libraries(dtln_probe PRIVATE Clarifae::TfLite m)
target_compile_options(dtln_probe PRIVATE -Wall -Wextra -Werror)
if(ANDROID)
 target_link_options(dtln_probe PRIVATE -Wl,-z,max-page-size=16384 -Wl,--no-undefined)
endif()
set(DTLN_TEST_MODEL_DIR "" CACHE PATH "Optional isolated test model directory; production default is vendor")
if(DTLN_TEST_MODEL_DIR)
 target_compile_definitions(dtln_probe PRIVATE
  CLARIFAE_DTLN_MODEL1_PATH="${DTLN_TEST_MODEL_DIR}/model_1.tflite"
  CLARIFAE_DTLN_MODEL2_PATH="${DTLN_TEST_MODEL_DIR}/model_2.tflite")
endif()
